+ plainText: string,
+ storedHash?: string | null
+): Promise<boolean> {
+ if (!storedHash) return false;
+
+ const enteredText = (plainText || '').trim();

+
+ // 1. Bcrypt hashes ($2a$, $2b$, $2y$): real verification, fail closed.
+ if (/^\$2[aby]\$/.test(storedHash)) {

+ return false;

+
+ // 3. SHA-256 hash
